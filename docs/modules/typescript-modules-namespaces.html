CommonJS 模块

Node.js 模块系统

require / module.exports 常用 
// 配置 module: "commonjs"
// TypeScript 自动转换 import/export 为 require/exports

// TypeScript 写法
import { add } from './math'
export function add(a: number, b: number) { return a + b }

// 编译后的 JavaScript
const { add } = require('./math')
function add(a, b) { return a + b }
module.exports = { add }

// 默认导出
export default class Logger {
  log(message: string) {
    console.log(message)
  }

// 默认导出函数
export default function greet(name: string): string {
  return `Hello, ${name}`

// 导入
import Logger from './logger'
import greet from './utils'

// 混合导出
export default class Utils {
  static format(date: Date): string {
    return date.toISOString()
  }
}

export function sleep(ms: number): Promise<void> {
  return new Promise(resolve => setTimeout(resolve, ms))
}

// 导入
import Utils, { sleep } from './utils'

命名空间（Namespace）

组织相关声明

namespace 进阶 
namespace Validation {
  export interface Validator {
    validate(value: string): boolean
  }

  export class EmailValidator implements Validator {
    validate(value: string): boolean {
      return value.includes('@')
    }
  }

  export class PhoneValidator implements Validator {
    validate(value: string): boolean {
      return /^\d{11}$/.test(value)
    }
  }
}

// 使用
const emailValidator = new Validation.EmailValidator()
console.log(emailValidator.validate('test@example.com'))  // true

命名空间合并

声明合并

Declaration Merging 进阶 
// 类与命名空间合并
class Greeter {
  greeting: string
  constructor(message: string) {
    this.greeting = message
  }
}

namespace Greeter {
  export function create(name: string): Greeter {
    return new Greeter(`Hello, ${name}`)
  }
}

// 使用
const greeter = Greeter.create('Alice')

// 函数与命名空间合并
function buildLabel(name: string): string {
  return `Label: ${name}`
}

namespace buildLabel {
  export let count = 0
}

buildLabel.count++  // 合并后可以访问

环境声明（declare）

为 JS 代码添加类型

declare var/function/class/module/namespace 核心 
// declare const/let/var
declare const API_URL: string
declare let cache: Map<string, any>

// declare function
declare function fetch(url: string): Promise<Response>
declare function setTimeout(callback: () => void, ms: number): number

// declare class
declare class EventEmitter {
  on(event: string, callback: () => void): void
  emit(event: string): void
}

// declare module
declare module 'some-module' {
  export function doSomething(): void
  export const version: string
}

// declare namespace
declare namespace Express {
  interface Request {
    body: any
  }
  interface Response {
    json(data: any): void
  }
}

类型定义文件

.d.ts 核心 
// globals.d.ts - 全局类型
declare const APP_NAME: string

// types/custom.d.ts - 模块声明
declare module 'custom-module' {
  export function doSomething(): void
  export interface Config {
    debug: boolean
    port: number
  }
}

// types/express.d.ts - 扩展已有类型

真实场景中的应用

案例 1：工具函数库 常用 
// utils/math.ts
export function add(a: number, b: number): number {
  return a + b
}

export function multiply(a: number, b: number): number {
  return a * b
}

// utils/string.ts
export function capitalize(str: string): string {
  return str.charAt(0).toUpperCase() + str.slice(1)
}

// utils/index.ts（统一入口）
export * from './math'
export * from './string'

// app.ts
import { add, capitalize } from './utils'

案例 3：模块与命名空间混合 进阶 
// validators.ts
export class EmailValidator {
  validate(email: string): boolean {
    return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)
  }
}

// 同时使用命名空间扩展
export namespace Validators {
  export class PhoneValidator {
    validate(phone: string): boolean {
      return /^1[3-9]\d{9}$/.test(phone)
    }
  }

  export class IdCardValidator {
    validate(idCard: string): boolean {
      return /^\d{17}[\dX]$/.test(idCard)
    }
  }
}
