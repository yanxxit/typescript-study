基础语法

定义和使用类型别名

type 核心 类型别名 - 给类型起新名字

Type 是类型的别名，可以给任何类型起一个新名字，包括原始类型、对象类型、联合类型等。

原始类型
string | number | boolean
→ type →
类型别名
ID = string | number
示例代码 
// 给原始类型起别名
type ID = string | number
type Callback = () => void

// 使用
let userId: ID = 123
let onClick: Callback = () => console.log('clicked')

// 对象类型
type User = {
  id: number
  name: string
  email: string
}

// 泛型类型
type Box<T> = {
  value: T
}

type NumberBox = Box<number>  // { value: number }

联合类型

使用 | 表示"或"关系

A | B 核心 联合类型 - 值可以是 A 或 B
string | number
string 或 number
→
值
'hello' 或 42
示例代码 
// 基本联合类型
type ID = string | number

let userId: ID = 123    // ✅
userId = 'abc'          // ✅

// 字面量联合类型
type Direction = 'up' | 'down' | 'left' | 'right'
type HttpStatus = 200 | 301 | 404 | 500

// 使用字面量联合
function move(direction: Direction) {
  console.log(`向${direction}移动`)
}

move('up')    // ✅
// move('forward')  // ❌ 不在联合类型中
可辨识联合 进阶 使用共同字段区分不同类型
示例代码 
type Shape =
  | { kind: 'circle'; radius: number }
  | { kind: 'rectangle'; width: number; height: number }
  | { kind: 'triangle'; base: number; height: number }

function area(shape: Shape): number {
  switch (shape.kind) {
    case 'circle':
      return Math.PI * shape.radius ** 2
    case 'rectangle':
      return shape.width * shape.height
    case 'triangle':
      return (shape.base * shape.height) / 2
  }
}

交叉类型

使用 & 表示"且"关系

A & B 核心 交叉类型 - 同时是 A 和 B
Person
name, age
→ & →
Employee
company, position
→
结果
name, age, company, position
示例代码 
type Person = {
  name: string
  age: number
}

type Employee = {
  company: string
  position: string
}

// 交叉类型：合并两个类型
type EmployeePerson = Person & Employee

// 等同于：
// {
//   name: string
//   age: number
//   company: string
//   position: string
// }

const staff: EmployeePerson = {
  name: 'Alice',
  age: 30,
  company: 'Tech Corp',
  position: 'Engineer'
}

元组类型

固定长度和类型的数组

[Type1, Type2, ...] 核心 元组 - 固定结构的数组
元组类型
[string, number]
→ 约束 →
值
['Alice', 30]
示例代码 
// 基本元组
type Point = [number, number]

const point: Point = [10, 20]  // ✅
// const point2: Point = [10, 20, 30]  // ❌ 长度不对
// const point3: Point = [10, '20']    // ❌ 类型不对

// 可选元素的元组
type HttpResponse = [number, string, string?]

const ok: HttpResponse = [200, 'OK']           // ✅
const notFound: HttpResponse = [404, 'Not Found', 'Page not found']  // ✅

// 只读元组
type readonlyPoint = readonly [number, number]

const rPoint: readonlyPoint = [10, 20]
// rPoint[0] = 30  // ❌ 只读元组不能修改

映射类型

遍历键创建新类型

{ [K in keyof T]: V } 进阶 映射类型 - 基于已有类型创建新类型

映射类型工作原理

flowchart LR
    A["User
id: number
name: string"] --> B["映射 + 修改"]
    B --> C["PartialUser
id?: number
name?: string"]
示例代码 
type User = {
  id: number
  name: string
  email: string
}

// 将所有属性变为可选
type PartialUser = {
  [K in keyof User]?: User[K]
}

// 将所有属性变为只读
type ReadonlyUser = {
  readonly [K in keyof User]: User[K]
}

// 内置映射类型
type Partial<T> = { [K in keyof T]?: T[K] }
type Required<T> = { [K in keyof T]-?: T[K] }
type Readonly<T> = { readonly [K in keyof T]: T[K] }

条件类型

根据条件选择类型

T extends U ? X : Y 进阶 条件类型 - 类型层面的三元运算
条件
T extends U ?
→
true → X
类型 X
false → Y
类型 Y
示例代码 
// 基本条件类型
type IsString<T> = T extends string ? true : false

type A = IsString<string>   // true
type B = IsString<number>   // false

// infer 关键字：推断类型
type ReturnType<T> = T extends (...args: any[]) => infer R ? R : never

type Fn = () => string
type Result = ReturnType<Fn>  // string

// 内置条件类型
type T1 = Exclude<'a' | 'b' | 'c', 'a'>  // 'b' | 'c'
type T2 = Extract<'a' | 'b' | 'c', 'a' | 'f'>  // 'a'

模板字面量类型

字符串级别的类型操作

`${A}-${B}` 进阶 模板字面量 - 编译时字符串拼接
示例代码 
// 基本模板字面量
type Color = 'red' | 'blue' | 'green'
type Size = 'sm' | 'md' | 'lg'

type ColorSize = `${Color}-${Size}`
// 'red-sm' | 'red-md' | 'red-lg' |
// 'blue-sm' | 'blue-md' | 'blue-lg' |
// 'green-sm' | 'green-md' | 'green-lg'

// 配合 Uppercase 等工具
type EventName<T extends string> = `on${Capitalize<T>}`

type ClickEvent = EventName<'click'>     // 'onClick'
type FocusEvent = EventName<'focus'>     // 'onFocus'

// 实际应用：CSS 类名
type Margin = '0' | '1' | '2' | '3' | '4' | 'auto'
type MarginClass = `m-${Margin}`
// 'm-0' | 'm-1' | 'm-2' | 'm-3' | 'm-4' | 'm-auto'

特性对比
特性 Type Interface

映射类型 ✅ ❌
条件类型 ✅ ❌

案例 2：深层 Partial 进阶 
// 递归将所有属性变为可选
type DeepPartial<T> = {
  [K in keyof T]?: T[K] extends object ? DeepPartial<T[K]> : T[K]
}

interface Config {
  db: {
    host: string
    port: number
    credentials: {
      username: string
      password: string
    }
  }
}

// 所有属性（包括嵌套）都变为可选
type PartialConfig = DeepPartial<Config>

const config: PartialConfig = {
  db: {
    host: 'localhost'
    // 其他都可选
  }
}
案例 3：类型安全的事件系统 进阶 
type EventMap = {
  click: { x: number; y: number }
  keydown: { key: string }
  resize: { width: number; height: number }
}

type EventKey = keyof EventMap
type EventData<K extends EventKey> = EventMap[K]

class TypedEventEmitter {
  private listeners = new Map<string, Function[]>()

  on<K extends EventKey>(
    event: K,
    listener: (data: EventData<K>) => void
  ) {
    const list = this.listeners.get(event) || []
    list.push(listener)
    this.listeners.set(event, list)
  }

  emit<K extends EventKey>(event: K, data: EventData<K>) {
    this.listeners.get(event)?.forEach(fn => fn(data))
  }
}

const emitter = new TypedEventEmitter()
emitter.on('click', (data) => {
  console.log(data.x, data.y)  // 类型安全
})
