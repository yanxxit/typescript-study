as 语法

推荐的类型断言方式

x as Type 核心 类型断言 - 告诉编译器类型 
// 基本用法
let a: any = 'hello'
let b = a as string
console.log(b.toUpperCase())  // ✅

// DOM 操作
const div = document.getElementById('app') as HTMLDivElement
div.style.color = 'red'

// 与类型注解的区别
let x: string = 'hello'    // 类型注解（编译器检查）
let y = 'hello' as string  // 类型断言（不检查）

// 尖括号语法（不推荐，与 JSX 冲突）
let length = (<string>someValue).length

as const 断言

固定字面量类型

x as const 核心 as const - 生成最精确的只读类型
❌ 没有 as const
{ apiUrl: string; timeout: number }
→ as const →
✅ 使用 as const
{ readonly apiUrl: 'https://...'; readonly timeout: 5000 }
// 对象
const config = {
  apiUrl: 'https://api.example.com',
  timeout: 5000
} as const
// { readonly apiUrl: 'https://api.example.com'; readonly timeout: 5000 }

// 数组
const arr = ['a', 'b', 'c'] as const
// readonly ['a', 'b', 'c']

// 替代枚举
const STATUS = {
  PENDING: 'pending',
  ACTIVE: 'active',
  DELETED: 'deleted'
} as const

type Status = typeof STATUS[keyof typeof STATUS]
// 'pending' | 'active' | 'deleted'

非空断言（!）

排除 null/undefined

x! 常用 非空断言 - 假设值不为 null 
// 可能为 null
function getLength(value: string | null): number {
  // return value.length  // ❌ 编译器报错
  return value!.length  // ✅ 使用非空断言
}

// DOM 操作
const element = document.getElementById('app')!
element.style.color = 'red'

// ⚠️ 非空断言不会在运行时检查
let value: string | null = null
console.log(value!.length)  // ❌ 运行时错误！

// 更安全的写法
function process(value: string | null): string {
  if (value === null) {
    return 'default'
  }
  return value.length.toString()  // ✅ 编译器自动缩窄
}

类型断言的限制

不能随意断言

类型断言规则 注意 
// ❌ 不允许任意类型间断言
let num: number = 42
let str = num as string  // ❌ 报错

// ✅ 使用 unknown 中转（双重断言）
let str2 = num as unknown as string  // ✅

// ✅ 可以从联合类型断言为具体类型
let value: string | number = 'hello'
let str3 = value as string  // ✅

// ❌ 不允许具体类型断言为联合类型
let str4: string = 'hello'
let union = str4 as string | number  // ❌ 报错

类型推断

编译器自动推导

TypeScript Inference 核心 类型推断 - 编译器自动推导类型 
// 变量推断
let name = 'Alice'      // string
let age = 30             // number
let active = true        // boolean
let arr = [1, 2, 3]  // number[]

// 函数返回值推断
function add(a: number, b: number) {
  return a + b  // 自动推断为 number
}

// 回调参数推断
const numbers = [1, 2, 3]
numbers.map(x => x * 2)  // x 自动推断为 number

// 最佳通用类型
let mixed = [1, 'hello', true]
// 推断为 (number | string | boolean)[]

infer 关键字

条件类型中推断类型

infer T 进阶 infer - 在条件类型中推断类型

infer 工作原理

flowchart LR
    A["T extends (...args: any[]) => infer R"] --> B{"T 是函数?"}
            B -->|Yes| C["返回 R（推断的返回类型）"]
            B -->|No| D["返回 never"]
          
// 推断函数返回类型
type ReturnType<T> = T extends (...args: any[]) => infer R ? R : never

type Fn = () => string
type Result = ReturnType<Fn>  // string

// 推断 Promise 内部类型
type UnwrapPromise<T> = T extends Promise<infer R> ? R : T

type A = UnwrapPromise<Promise<string>>           // string
type B = UnwrapPromise<Promise<Promise<number>>>  // number

// 推断函数参数类型
type FirstArg<T> = T extends (first: infer F, ...args: any[]) => any ? F : never

type C = FirstArg<(name: string, age: number) => void>  // string

// 推断元组类型
type Head<T> = T extends [infer H, ...any[]] ? H : never
type Tail<T> = T extends [any, ...infer R] ? R : never

type D = Head<[1, 2, 3]>  // 1
type E = Tail<[1, 2, 3]>  // [2, 3]

实战案例

真实场景中的应用

案例 1：DOM 操作 常用 
// 不推荐：使用断言
const div = document.getElementById('app') as HTMLDivElement

// 推荐：使用类型守卫
function getElementById(id: string): HTMLElement {
  const element = document.getElementById(id)
  if (!element) {
    throw new Error(`元素 ${id} 不存在`)
  }
  return element  // 类型缩窄，无需断言
}

// 确定类型时使用断言
const input = document.getElementById('myInput') as HTMLInputElement
input.value = 'hello'
案例 2：API 响应处理 常用 
interface User {
  id: number
  name: string
  email: string
}

// 类型守卫
function isUser(value: unknown): value is User {
  return (
    typeof value === 'object' &&
    value !== null &&
    'id' in value &&
    'name' in value &&
    'email' in value
  )
}

async function fetchUser(id: number): Promise<User> {
  const response = await fetch(`/api/users/${id}`)
  const data = await response.json()
  if (!isUser(data)) {
    throw new Error('无效的用户数据')
  }
  return data  // 类型安全
}
案例 3：泛型工具函数 进阶 
// 使用 infer 推断类型
type UnwrapPromise<T> = T extends Promise<infer R> ? R : T
type DeepUnwrap<T> = T extends Promise<infer R> ? DeepUnwrap<R> : T

// 使用示例
async function fetchData() {
  return { id: 1, name: 'Alice' }
}

type DataType = DeepUnwrap<ReturnType<typeof fetchData>>
// { id: number; name: string }
案例 4：as const 模拟枚举 常用 
// 使用 as const 替代枚举
const Directions = {
  UP: 'up',
  DOWN: 'down',
  LEFT: 'left',
  RIGHT: 'right'
} as const

type Direction = typeof Directions[keyof typeof Directions]
// 'up' | 'down' | 'left' | 'right'

// 使用
function move(direction: Direction) {
  console.log(`向${direction}移动`)
}

move(Directions.UP)  // ✅ 自动补全
move('up')           // ✅ 也支持字符串
