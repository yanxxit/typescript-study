原始类型

JavaScript 的 7 种基本类型

string

字符串，支持单引号、双引号、模板字符串

number

数字，包括整数、浮点数、科学计数法

boolean

特殊类型

TypeScript 独有的类型

any 慎用 任意类型 - 绕过类型检查 
let anything: any = 'hello'
anything = 42        // ✅
anything = true      // ✅
anything.foo()       // ✅ 不报错，但运行时可能出错
unknown 推荐 未知类型 - 类型安全的 any
any
不做任何检查
vs
unknown
必须检查后使用
let value: unknown = 'hello'

// value.toUpperCase()  // ❌ 错误
if (typeof value === 'string') {
  value.toUpperCase()   // ✅ 正确
}
never 常用 永不存在的值 
// 抛出错误
function throwError(msg: string): never {
  throw new Error(msg)
}

// 无限循环
function infiniteLoop(): never {
  while (true) {}
}

// 穷尽检查
type Shape = { kind: 'circle' } | { kind: 'square' }
function area(s: Shape) {
  switch (s.kind) {
    case 'circle': return 1
    case 'square': return 2
    default: const _exhaustive: never = s
  }
}
void 常用 无返回值 
function log(message: string): void {
  console.log(message)
}

联合与交叉类型

组合多个类型

A | B / A & B 核心
联合 A | B
A 或 B
vs
交叉 A & B
A 且 B
// 联合类型：值可以是多种类型之一
type ID = string | number
type Direction = 'up' | 'down' | 'left' | 'right'

// 交叉类型：合并多个类型
type Person = { name: string; age: number }
type Employee = { company: string }
type Staff = Person & Employee
// { name: string; age: number; company: string }

字面量类型

值本身就是类型

'value' | 123 | true 核心 
// 字符串字面量
type Theme = 'light' | 'dark'
type Direction = 'up' | 'down' | 'left' | 'right'

// 数字字面量
type DiceValue = 1 | 2 | 3 | 4 | 5 | 6

// const 断言：让对象变成只读字面量
const config = {
  host: 'localhost',
  port: 3000
} as const
// 类型：{ readonly host: 'localhost'; readonly port: 3000 }

函数类型

函数签名、重载、泛型

(a: number) => string 核心 
// 函数类型
type MathFn = (a: number, b: number) => number

// 可选参数
function greet(name: string, greeting?: string): string {
  return `${greeting || 'Hello'}, ${name}`
}

// 剩余参数
function sum(...numbers: number[]): number {
  return numbers.reduce((a, b) => a + b, 0)
}

// 函数重载
function format(value: string): string
function format(value: number): string
function format(value: string | number): string {
  if (typeof value === 'string') return value.toUpperCase()
  return value.toFixed(2)
}

类类型

访问修饰符、抽象类、接口实现

class / abstract class 核心 
class User {
  public name: string      // 公开（默认）
  private age: number      // 私有
  protected email: string  // 受保护
  readonly id: number      // 只读

  constructor(name: string, age: number, email: string, id: number) {
    this.name = name
    this.age = age
    this.email = email
    this.id = id
  }
}

// 抽象类
abstract class Shape {
  abstract area(): number

  describe(): string {
    return `面积: ${this.area()}`
  }
}

// 接口实现
interface Printable {
  print(): void
}

class Document implements Printable {
  print() { console.log('打印') }
}

命名常量集合

  Right    // 3
}

// 字符串枚举

  Light = 'light',
  Dark = 'dark'
}

// const 枚举：编译时内联

  Red,
  Green,
  Blue
}
const c = Color.Red  // 编译后：const c = 0

泛型

参数化类型

<T> 核心 
// 泛型函数
function identity<T>(value: T): T {
  return value
}

// 泛型约束
function getProperty<T, K extends keyof T>(obj: T, key: K): T[K] {
  return obj[key]
}

// 泛型类
class Container<T> {
  private items: T[] = []
  add(item: T) { this.items.push(item) }
  get(index: number): T { return this.items[index] }
}

高级类型

条件类型、映射类型、模板字面量

条件类型 / 映射类型 / 模板字面量 进阶 
// 条件类型
type IsString<T> = T extends string ? true : false

// infer 推断
type ReturnType<T> = T extends (...args: any[]) => infer R ? R : never

// 映射类型
type Partial<T> = { [K in keyof T]?: T[K] }
type Readonly<T> = { readonly [K in keyof T]: T[K] }

// 模板字面量类型
type EventName<T extends string> = `on${Capitalize<T>}`
type ClickEvent = EventName<'click'>  // 'onClick'

类型守卫与断言

运行时类型检查

typeof / instanceof / in / is 核心 
// typeof 守卫
function format(value: string | number): string {
  if (typeof value === 'string') return value.toUpperCase()
  return value.toFixed(2)
}

// instanceof 守卫
function process(error: Error) {
  if (error instanceof TypeError) { /* ... */ }
}

// 自定义类型守卫
function isString(value: unknown): value is string {
  return typeof value === 'string'
}

// 类型断言
let value: unknown = 'hello'
let length = (value as string).length
