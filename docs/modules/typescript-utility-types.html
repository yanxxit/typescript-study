什么是 Utility Types

类型转换的基石

Utility Types
原始类型
User { id, name, email }
→ Utility Type →
变换后
PartialUser / ReadonlyUser / ...
interface User {
  id: number
  name: string
  email: string
}

// 使用 Utility Types 变换类型
type PartialUser = Partial<User>        // 所有属性可选
type ReadonlyUser = Readonly<User>      // 所有属性只读
type UserName = Pick<User, 'name'>     // 只取 name
type SafeUser = Omit<User, 'email'>  // 排除 email

对象操作

变换对象类型的工具

Partial<Type> 核心 部分类型 - 所有属性可选 
type Partial<T> = { [K in keyof T]?: T[K] }

// 使用
function updateTodo(todo: Todo, fields: Partial<Todo>): Todo {
  return { ...todo, ...fields }
}
Required<Type> 核心 必选类型 - 所有属性必选 
type Required<T> = { [K in keyof T]-?: T[K] }

// 使用
function createServer(config: Required<Config>) {
  console.log(`服务器: ${config.host}:${config.port}`)
}
Pick<Type, Keys> 核心 选取类型 - 选取指定键 
type Pick<T, K extends keyof T> = { [P in K]: T[P] }

// 使用
type UserPreview = Pick<User, 'id' | 'name'>
// { id: number; name: string }
Omit<Type, Keys> 核心 排除类型 - 排除指定键 
type Omit<T, K extends keyof any> = Pick<T, Exclude<keyof T, K>>

// 使用
type SafeUser = Omit<User, 'password'>
// { id: number; name: string; email: string }
Record<Keys, Type> 核心 记录类型 - 构造键值映射 
type Record<K extends keyof any, T> = { [P in K]: T }

// 使用
const cats: Record<CatName, { age: number }> = {
  miffy: { age: 10 },
  boris: { age: 5 }
}

联合类型操作

过滤和提取联合类型

Exclude / Extract 核心
Exclude
'a' | 'b' | 'c' 排除 'a'
→
结果
'b' | 'c'
// Exclude: 排除
type T0 = Exclude<'a' | 'b' | 'c', 'a'>  // 'b' | 'c'

// Extract: 提取
type T1 = Extract<'a' | 'b' | 'c', 'a' | 'f'>  // 'a'

// NonNullable: 排除 null/undefined
type T2 = NonNullable<string | null | undefined>  // string

函数操作

提取函数类型信息

Parameters / ReturnType 核心 
// Parameters: 提取参数类型
function greet(name: string, age: number): void {}
type Params = Parameters<typeof greet>  // [string, number]

// ReturnType: 提取返回类型
function getUser() { return { id: 1, name: 'Alice' } }
type User = ReturnType<typeof getUser>  // { id: number; name: string }

// ConstructorParameters: 构造函数参数
class Article {
  constructor(title: string, content?: string) {}
}
type Params = ConstructorParameters<typeof Article>  // [string, string?]

// InstanceType: 实例类型
class Cat { name: string }
type CatInstance = InstanceType<typeof Cat>  // Cat

字符串操作

编译时字符串转换

Uppercase / Lowercase / Capitalize / Uncapitalize 核心 
type T1 = Uppercase<'hello'>      // 'HELLO'
type T2 = Lowercase<'HELLO'>      // 'hello'
type T3 = Capitalize<'hello'>     // 'Hello'
type T4 = Uncapitalize<'Hello'>   // 'hello'

// 配合模板字面量类型
type EventName<T extends string> = `on${Capitalize<T>}`
type ClickEvent = EventName<'click'>  // 'onClick'

类型推断工具

infer 和 Awaited

Awaited<Type> / infer 进阶 
// Awaited: 解包 Promise
type A = Awaited<Promise<string>>           // string
type B = Awaited<Promise<Promise<number>>>  // number

// infer: 条件类型中推断类型
type ReturnType<T> = T extends (...args: any[]) => infer R ? R : never
type FirstArg<T> = T extends (first: infer F, ...args: any[]) => any ? F : never

自定义 Utility Types

按需创建工具类型

DeepPartial / DeepReadonly / Nullable 进阶 
// DeepPartial: 深层可选
type DeepPartial<T> = {
  [K in keyof T]?: T[K] extends object ? DeepPartial<T[K]> : T[K]
}

// DeepReadonly: 深层只读
type DeepReadonly<T> = {
  readonly [K in keyof T]: T[K] extends object ? DeepReadonly<T[K]> : T[K]
}

// Nullable: 添加 null
type Nullable<T> = T | null

实战案例

真实场景中的应用

案例 1：类型安全的 API 常用 
interface User {
  id: number
  name: string
  email: string
  password: string
}

// 创建 DTO：排除敏感字段
type CreateUserDTO = Omit<User, 'id' | 'password'>

// 更新 DTO：部分字段可选
type UpdateUserDTO = Partial<Omit<User, 'id'>>

// 响应类型：排除密码
type UserResponse = Omit<User, 'password'>
案例 2：事件系统 常用 
type EventMap = {
  click: { x: number; y: number }
  keydown: { key: string }
  resize: { width: number; height: number }
}

type EventKey = keyof EventMap
type EventData<K extends EventKey> = EventMap[K]

function on<K extends EventKey>(
  event: K,
  handler: (data: EventData<K>) => void
) {
  // 类型安全的事件处理
}
