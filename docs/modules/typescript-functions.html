基本语法

函数声明与箭头函数

function / () => 核心 函数 - 两种写法
函数声明
function add(a, b) { return a+b }
vs
箭头函数
const add = (a, b) => a+b
示例代码 
// 函数声明
function add(a: number, b: number): number {
  return a + b
}

// 函数表达式
const add2 = function(a: number, b: number): number {
  return a + b
}

// 箭头函数
const add3 = (a: number, b: number): number => a + b

// 单行返回可省略
const double = (x: number) => x * 2

// 单参数可省略括号
const triple = x => x * 3

函数类型

定义函数的类型

(a: T) => R 核心 函数类型 - 类型别名 
// 函数类型别名
type MathFn = (a: number, b: number) => number

const add: MathFn = (a, b) => a + b
const subtract: MathFn = (a, b) => a - b

// 接口定义函数
interface SearchFunc {
  (source: string, subString: string): boolean
}

const search: SearchFunc = (source, sub) => {
  return source.includes(sub)
}

// 返回类型推断
function add(a: number, b: number) {
  return a + b  // 自动推断返回 number
}

可选参数

使用 ? 标记

x?: T 核心 
function greet(name: string, greeting?: string): string {
  return `${greeting || 'Hello'}, ${name}`
}

greet('Alice')           // 'Hello, Alice'
greet('Alice', 'Hi')     // 'Hi, Alice'

// ⚠️ 可选参数必须在必选参数之后
// function f(a?: string, b: number) {}  // ❌ 错误
function f(a: string, b?: number) {}  // ✅ 正确

默认参数

参数默认值

x = defaultValue 核心 
function greet(name: string, greeting: string = 'Hello'): string {
  return `${greeting}, ${name}`
}

greet('Alice')           // 'Hello, Alice'
greet('Alice', 'Hi')     // 'Hi, Alice'

// 默认值类型推断
function createArray<T>(length: number, value: T): T[] {
  return Array(length).fill(value)
}

createArray(3, 'hello')  // string[]
createArray(3, 42)       // number[]

剩余参数

收集不定数量的参数

...args: T[] 核心 
function sum(...numbers: number[]): number {
  return numbers.reduce((a, b) => a + b, 0)
}

sum(1, 2, 3)        // 6
sum(1, 2, 3, 4, 5)  // 15

// 混合使用
function log(level: string, ...messages: string[]): void {
  console.log(`[${level}]`, ...messages)
}

log('INFO', 'Server', 'started')  // [INFO] Server started

函数重载

为同一函数定义多个签名

function overload 进阶 
// 基本重载
function format(value: string): string
function format(value: number): string
function format(value: string | number): string {
  if (typeof value === 'string') {
    return value.toUpperCase()
  }
  return value.toFixed(2)
}

format('hello')  // 'HELLO'
format(3.14159)  // '3.14'

// 复杂重载
function createElement(tag: 'div'): HTMLDivElement
function createElement(tag: 'span'): HTMLSpanElement
function createElement(tag: string): HTMLElement {
  return document.createElement(tag)
}

const div = createElement('div')    // HTMLDivElement
const span = createElement('span')  // HTMLSpanElement

this 参数

显式指定 this 类型

this: Type 常用 
interface User {
  name: string
  greet(this: User): string
}

const user: User = {
  name: 'Alice',
  greet() {
    return `Hello, ${this.name}`
  }
}

user.greet()  // ✅

// const fn = user.greet
// fn()  // ❌ this 丢失

// 函数中的 this 参数
function greet(this: { name: string }) {
  return `Hello, ${this.name}`
}

const obj = { name: 'Alice', greet }
obj.greet()  // ✅

异步函数

async/await

async function / await 核心

异步函数流程

flowchart LR
    A["async function"] --> B["await 异步操作"]
            B --> C["返回 Promise"]
            C --> D["获取结果"]
          
// async/await
async function fetchUser(id: number): Promise<User> {
  const response = await fetch(`/api/users/${id}`)
  const user: User = await response.json()
  return user
}

// 泛型异步函数
async function fetchData<T>(url: string): Promise<T> {
  const response = await fetch(url)
  return response.json()
}

// 错误处理
async function safeFetch(url: string): Promise<Response | null> {
  try {
    return await fetch(url)
  } catch (error) {
    console.error('Fetch failed:', error)
    return null
  }
}

高阶函数

函数作为参数或返回值

Higher-Order Function 进阶 
// 函数作为参数
function map<T, U>(array: T[], fn: (item: T) => U): U[] {
  return array.map(fn)
}

const numbers = [1, 2, 3]
const doubled = map(numbers, x => x * 2)  // [2, 4, 6]

// 返回函数
function createGreeter(greeting: string) {
  return function (name: string) {
    return `${greeting}, ${name}`
  }
}

const sayHello = createGreeter('Hello')
sayHello('Alice')  // 'Hello, Alice'

// 装饰器风格
function withLogging<T extends (...args: any[]) => any>(fn: T): T {
  return function (...args: any[]) {
    console.log(`调用 ${fn.name}`)
    const result = fn(...args)
    console.log(`结果: ${JSON.stringify(result)}`)
    return result
  } as T
}

案例 1：类型安全的事件系统 常用 
type EventMap = {
  click: { x: number; y: number }
  keydown: { key: string }
  resize: { width: number; height: number }
}

type EventKey = keyof EventMap

function on<K extends EventKey>(
  event: K,
  handler: (data: EventMap[K]) => void
) {
  // 注册事件处理
}

on('click', (data) => {
  console.log(data.x, data.y)  // 类型安全
})

interface ApiResponse<T> {
  code: number
  data: T
  message: string
}

async function apiGet<T>(url: string): Promise<ApiResponse<T>> {
  const response = await fetch(url)
  return response.json()
}

interface User {
  id: number
  name: string
}

const result = await apiGet<User>('/api/user/1')
console.log(result.data.name)  // 类型安全
案例 3：函数组合 进阶 
function compose<A, B, C>(
  f: (b: B) => C,
  g: (a: A) => B
): (a: A) => C {
  return (a) => f(g(a))
}

const add10 = (x: number) => x + 10
const double = (x: number) => x * 2

const add10ThenDouble = compose(double, add10)
add10ThenDouble(5)  // 30
案例 4：柯里化 进阶 
function curry<A extends any[], B, C>(
  fn: (...args: [...A, B]) => C
): (...args: A) => (arg: B) => C {
  return (...args: A) => (arg: B) => fn(...args, arg)
}

const add = (a: number, b: number) => a + b
const add5 = curry(add)(5)

add5(3)   // 8
add5(10)  // 15
